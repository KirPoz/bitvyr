function(bitvyr_test name source scalar)
    add_executable(${name} ${source})
    if(scalar)
        target_link_libraries(${name} PRIVATE nes_scalar_reference)
    else()
        target_link_libraries(${name} PRIVATE NES::Core)
    endif()
    nes_native_test_options(${name})
endfunction()

# CPU/PPU остаются scalar; FIR/phase quality совпадает с selected продуктом.
# Существующий nes_scalar_reference и его ожидаемое PCM-различие не меняются.
bitvyr_create_core(nes_fastcap_scalar_audio "${core_root}" TRUE)
nes_native_test_options(nes_fastcap_scalar_audio)
get_target_property(fastcap_selected_definitions nes_core INTERFACE_COMPILE_DEFINITIONS)
get_target_property(fastcap_own_definitions nes_core COMPILE_DEFINITIONS)
list(APPEND fastcap_selected_definitions ${fastcap_own_definitions})
list(FILTER fastcap_selected_definitions INCLUDE REGEX "^(NES_CORE_APU_|NES_CORE_PORTABLE_NATIVE_TRIANGLE_MEAN=)")
list(REMOVE_DUPLICATES fastcap_selected_definitions)
if(NOT "NES_CORE_APU_DENSE_PHASES=1" IN_LIST fastcap_selected_definitions OR
   NOT "NES_CORE_APU_HYBRID_RANK4=1" IN_LIST fastcap_selected_definitions)
    message(FATAL_ERROR "Scalar audio oracle requires actual selected DENSE_PHASES and HYBRID_RANK4 definitions")
endif()
target_compile_definitions(nes_fastcap_scalar_audio PUBLIC ${fastcap_selected_definitions})
add_executable(mapper_expansion_fastcap_scalar_audio MapperExpansionFastcapTests.cpp)
target_link_libraries(mapper_expansion_fastcap_scalar_audio PRIVATE nes_fastcap_scalar_audio)
target_compile_definitions(mapper_expansion_fastcap_scalar_audio PRIVATE FASTCAP_SCALAR=1)
nes_native_test_options(mapper_expansion_fastcap_scalar_audio)
add_test(NAME mapper_expansion_fastcap_scalar_audio COMMAND mapper_expansion_fastcap_scalar_audio --canonical "${CMAKE_CURRENT_BINARY_DIR}/fastcap-scalar-audio.canonical")
set_tests_properties(mapper_expansion_fastcap_scalar_audio PROPERTIES FIXTURES_SETUP fastcap_scalar_audio_canonical)
